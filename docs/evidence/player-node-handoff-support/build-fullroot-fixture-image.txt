def inventory(path):
 result={};hardlinks={}
 with tarfile.open(path,'r:') as archive:
  for member in archive:
   name=member.name.removeprefix('./').rstrip('/')
   if name in ('','.'):continue
   assert name not in result and name not in hardlinks,name
   data={'mode':member.mode,'uid':member.uid,'gid':member.gid,'xattrs':{k:v for k,v in member.pax_headers.items() if '.xattr.' in k}}
   if member.isfile():
    data.update(kind='file',size=member.size,sha256=hashlib.file_digest(archive.extractfile(member),'sha256').hexdigest())
   elif member.issym():data.update(kind='link',target=member.linkname)
   elif member.isdir():data.update(kind='directory')
   elif member.islnk():hardlinks[name]=(member.linkname.removeprefix('./'),data);continue
   else:data.update(kind=member.type.decode(),major=member.devmajor,minor=member.devminor)
   result[name]=data
 while hardlinks:
  before=len(hardlinks)
  for name,(target,data) in list(hardlinks.items()):
   if target in result:
    result[name]={**result[target],**data};del hardlinks[name]
  assert len(hardlinks)<before,'unresolved hardlink'
 return result
