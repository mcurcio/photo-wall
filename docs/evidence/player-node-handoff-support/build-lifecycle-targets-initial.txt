PROD=ROOT/'components'
WORK=ROOT/'lifecycle-fixtures'
WORK.mkdir()
sys.path.insert(0,str(PROD/'source'))
from scripts.build_app_environment import build
from scripts.debian_packages import PIN
from scripts.node_build_inputs import BUILDER_IMAGE
ASSEMBLER='sha256:90c772d80b3d8a6c0b733e3fdd4464e14e83bc331dc757d828c0b4b8e57cef8c'
tempfile.tempdir=str(ROOT/'scratch')
