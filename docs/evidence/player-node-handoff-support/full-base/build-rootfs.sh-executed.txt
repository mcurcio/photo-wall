#!/bin/bash
# Source and components are read-only; /work is the new task-only native Linux volume.
set -euo pipefail
mkdir /work/rootfs-build-tmp
export TMPDIR=/work/rootfs-build-tmp
export PYTHONDONTWRITEBYTECODE=1
export PYTHONPATH=/source:/opt/photo-wall-build-python
[ "$(uname -m)" = aarch64 ]
[ "$(git -C /opt/rpi-image-gen rev-parse HEAD)" = 262d4df5a9f9d4133370465399a7958a7c22cdc7 ]
[ ! -e /work/photo-wall-base-work ]
python3 -B /recipes/prepare-rootfs.py
export SOURCE_DATE_EPOCH="$(python3 -B /source/scripts/debian_packages.py epoch)"
/opt/rpi-image-gen/rpi-image-gen build -S /source/appliance/rpi_image_gen -c photo-wall-base.yaml -- \
  IGconf_sys_workroot=/work/photo-wall-base-work \
  IGconf_app_bootstrapper_deb=/work/rootfs-inputs/bootstrapper.deb \
  IGconf_app_node_base_deb=/components/node-base.deb \
  IGconf_app_node_display_deb=/components/node-display.deb \
  IGconf_app_device_packages=/work/rootfs-inputs/device-packages.list \
  IGconf_app_debian_sources=/work/rootfs-inputs/debian-sources.list
# Exit 0 is only build completion. The workflow's independent content/device/ABI
# checks and later kernel/initrd/PXE assembly remain mandatory acceptance steps.
