"""Materialize exact production acceptance/build bodies with explicit local adapters."""
import hashlib
import json
from pathlib import Path
import yaml

source = Path('/source')
output = Path('/work/workflow-steps')
workflow = source / '.github/workflows/base-image.yml'
document = yaml.safe_load(workflow.read_text())
steps = {s.get('name'): s for job in document['jobs'].values() for s in job.get('steps', [])}
selected = {
    'verify-rootfs': 'Verify the built base image is what 0009 requires (content, not just build success)',
    'build-kernel': 'Build the scratch kernel + initrd root (Debian trixie arm64)',
    'verify-kernel-config': "Verify the cached kernel config still builds in stage 1's kernel support",
    'assemble-bundle': 'Assemble the netboot base bundle and content-verify the initrd',
    'verify-node-abi': 'Verify installed node ABI and produce the explicit V2 cohort bundle',
    'verify-bundle': 'Verify the assembled bundle (content, no hardware)',
}
output.mkdir()
record = {'workflow_sha256': hashlib.sha256(workflow.read_bytes()).hexdigest(), 'steps': {}}
for filename, name in selected.items():
    original = steps[name]['run']
    assert '${{' not in original
    rendered = original.replace('.venv/bin/python', 'python3')
    adaptations = ['Debian tool-image python3 replaces checkout .venv/bin/python']
    if filename == 'build-kernel':
        # Tools are already installed and inventoried in the pinned image. Preserve
        # every scratch/chroot/build action; omit only mutable HOST apt installation.
        begin = rendered.index('sudo apt-get update\n')
        end_marker = 'mmdebstrap debian-archive-keyring ca-certificates gnupg curl initramfs-tools\n'
        end = rendered.index(end_marker, begin) + len(end_marker)
        rendered = rendered[:begin] + '# Host dependencies verified in tool-packages.tsv.\n' + rendered[end:]
        adaptations.append('omitted host apt update/install only; scratch apt steps unchanged')
    target = output / (filename + '.sh')
    target.write_text('#!/bin/bash\n' + rendered)
    record['steps'][filename] = {
        'name': name,
        'original_sha256': hashlib.sha256(original.encode()).hexdigest(),
        'rendered_sha256': hashlib.sha256(target.read_bytes()).hexdigest(),
        'adaptations': adaptations,
    }
(output / 'provenance.json').write_text(json.dumps(record, indent=2, sort_keys=True) + '\n')
