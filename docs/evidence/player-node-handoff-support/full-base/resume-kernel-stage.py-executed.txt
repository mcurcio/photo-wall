"""Resume exact workflow selection/staging after successful mkinitramfs."""
from pathlib import Path
script = Path('/work/workflow-steps/build-kernel.sh').read_text()
start = script.index('python_libdir=$(sudo chroot')
end = script.index('if ! sudo chroot "$SCRATCH" mkinitramfs', start)
tail = script.index('# Locate the kernel image, Pi 5 DTB, and overlays')
Path('/work/workflow-steps/resume-kernel-stage.sh').write_text(
    '#!/bin/bash\nset -euo pipefail\n'
    'test -s "$SCRATCH/boot/initrd.img"\ntest ! -e "$KOUT"\n'
    'cmp /source/appliance/netboot_initramfs/hooks/photo-wall-netboot "$SCRATCH/etc/initramfs-tools/hooks/photo-wall-netboot"\n'
    'cmp /source/appliance/netboot_initramfs/scripts/photowall-netboot "$SCRATCH/etc/initramfs-tools/scripts/photowall-netboot"\n'
    + script[start:end] + script[tail:])
