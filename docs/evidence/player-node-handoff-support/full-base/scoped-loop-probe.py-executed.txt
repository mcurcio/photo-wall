"""Task-local authority restriction around the unchanged production mount probe."""
from pathlib import Path
import fcntl,hashlib,json,os,secrets,subprocess,sys
sys.path.insert(0,'/source')
from scripts import initrd_mount_probe as probe
from appliance.bootstrap import LOOP_INFO64
minor=int(os.environ['TASK_LOOP_MINOR'])
assert 0 <= minor <= 1048575
name='stage1-pw-'+secrets.token_hex(8)+'.squashfs'

def mappings():
    result={}
    for path in Path('/sys/block').glob('loop*/loop/backing_file'):
        result[path.parent.parent.name]=path.read_text().strip()
    return result

def redacted(data):
    return {key:hashlib.sha256(value.encode()).hexdigest() for key,value in data.items()}

before=mappings()
assert not before.get('loop'+str(minor)), 'selected_loop_already_bound'
assert all(Path(value).name != name for value in before.values()), 'probe_name_collision'
probe.PROBE_IMAGE=name
work=Path('/work/initrd-mount-probe-'+name.removesuffix('.squashfs'))
print(json.dumps({'selected_minor':minor,'probe_name':name,'before_backing_hashes':redacted(before)}),flush=True)
original_run=subprocess.run

def scoped_run(args,*pos,**kwargs):
    # Production cleanup is already basename-limited; enforce selected minor and
    # exact generated image basename again before granting any detach operation.
    if args[:2] == ['losetup','-d']:
        assert args[2] == '/dev/loop'+str(minor), 'detach_other_device_refused'
        current=mappings().get('loop'+str(minor),'')
        assert Path(current).name == name, 'detach_other_backing_refused'
        expected=(work/'root'/probe.PROBE_DIR/name).stat()
        with open(args[2], 'rb', buffering=0) as device:
            fields=LOOP_INFO64.unpack(fcntl.ioctl(device.fileno(),0x4C05,bytes(LOOP_INFO64.size)))
        assert fields[:2] == (expected.st_dev,expected.st_ino), 'detach_other_inode_refused'
    return original_run(args,*pos,**kwargs)

failure=None
try:
    violations=probe.probe(Path('/work/photo-wall-node-bundle/boot/initrd.img'),work,run=scoped_run)
except BaseException as error:
    failure=error
    violations=['probe_exception:'+type(error).__name__]
after=mappings()
if any(after.get(key) != value for key,value in before.items()):
    violations.append('preexisting_loop_mapping_changed')
if any(Path(value).name == name for value in after.values()):
    violations.append('task_loop_mapping_remains')
print(json.dumps({'after_backing_hashes':redacted(after),'violations':violations}),flush=True)
if failure is not None:
    raise failure
if violations:
    raise SystemExit(1)
print('SCOPED_REAL_INITRD_MOUNT_PROBE_PASS')
