"""Independently reopen frozen component archives and verify exact installed membership."""
import hashlib
import json
import subprocess
import tarfile
from pathlib import Path

WORK = Path('/Volumes/Dock/Temp/photo-wall-stop-c07666b/components')


def stream_sha(stream):
    digest = hashlib.sha256()
    while chunk := stream.read(1024 * 1024):
        digest.update(chunk)
    return digest.hexdigest()


def file_sha(path):
    with path.open('rb') as source:
        return stream_sha(source)


def main():
    components = json.loads((WORK / 'components.json').read_text())
    abi = components['abi']
    results = {'abi': abi, 'provenance': 'dirty-source components; no squashfs, PXE or physical qualification'}
    for role, field in [('app', 'app_environment'), ('manager-primary', 'manager_primary')]:
        reference = json.loads((WORK / (role + '-reference.json')).read_text())
        assert reference == components[field]
        assert {key: reference[key] for key in abi} == abi
        archive = WORK / (role + '.tar')
        assert file_sha(archive) == reference['environment_sha256']
        assert archive.stat().st_size == reference['size_bytes']
        assert file_sha(WORK / (role + '.deb')) == reference['deb_sha256']
        with tarfile.open(archive, 'r:') as opened:
            members = opened.getmembers()
            assert len({member.name for member in members}) == len(members)
            metadata = json.load(opened.extractfile('environment.json'))
            assert metadata['reference'] == {key: value for key, value in reference.items() if key not in ('environment_sha256', 'size_bytes')}
            actual = {}
            for member in members:
                if not member.name.startswith('rootfs/') or member.isdir():
                    continue
                name = member.name[len('rootfs/'):]
                if member.issym():
                    actual[name] = {'link': member.linkname}
                else:
                    assert member.isfile(), name
                    actual[name] = {'mode': member.mode, 'size': member.size, 'sha256': stream_sha(opened.extractfile(member))}
            assert actual == metadata['files']
            assert stream_sha(opened.extractfile('dependency-lock.json')) == reference['dependency_lock_sha256']
            assert stream_sha(opened.extractfile('sources.json')) == reference['source_snapshot_sha256']
            sources = json.load(opened.extractfile('sources.json'))
            for name, digest in sources['input_files'].items():
                assert file_sha(WORK / 'source' / name) == digest
            prefix = 'usr/lib/photo-wall-player/' if role == 'app' else 'usr/lib/photo-wall-node-manager/'
            matched = 0
            for name, entry in actual.items():
                if name.startswith(prefix) and name.endswith('.py'):
                    source = WORK / 'source' / name[len(prefix):]
                    if source.is_file():
                        assert file_sha(source) == entry['sha256']
                        matched += 1
            assert matched == (49 if role == 'app' else 23)
            placeholder = actual['etc/photo-wall/public.json']
            assert placeholder == {'mode': 0o444, 'size': 0, 'sha256': hashlib.sha256(b'').hexdigest()}
            if role == 'app':
                assert actual['usr/lib/photo-wall-client/libphoto-wall-frame-client.so']['sha256'] == file_sha(WORK / 'libphoto-wall-frame-client.so')
            results[role] = {'archive_sha256': reference['environment_sha256'], 'bytes': reference['size_bytes'],
                             'verified_members': len(actual), 'source_files': matched, 'capacity': metadata['capacity'],
                             'dependency_image': sources['built_image'], 'dependency_lock': reference['dependency_lock_sha256'],
                             'source_provenance': reference['source_snapshot_sha256']}
    # Inspect exact Debian installed payload rather than trusting the staged tree.
    process = subprocess.Popen(['docker', 'run', '--rm', '--network', 'none', '-v', str(WORK) + ':/work:ro',
                                'sha256:90c772d80b3d8a6c0b733e3fdd4464e14e83bc331dc757d828c0b4b8e57cef8c',
                                'dpkg-deb', '--fsys-tarfile', '/work/node-base.deb'], stdout=subprocess.PIPE)
    base_files = 0
    with tarfile.open(fileobj=process.stdout, mode='r|') as payload:
        for member in payload:
            name = member.name.removeprefix('./')
            if member.isdir():
                continue
            staged = WORK / 'base' / name
            if member.issym():
                assert staged.is_symlink() and str(staged.readlink()) == member.linkname
            else:
                assert member.isfile() and file_sha(staged) == stream_sha(payload.extractfile(member))
                assert member.mode == staged.stat().st_mode & 0o7777
                base_files += 1
    assert process.wait() == 0
    base_marker = json.loads((WORK / 'base/usr/lib/photo-wall-node-base/abi.json').read_text())
    assert base_marker['base_abi'] == abi['base_abi']
    control = (WORK / 'base/DEBIAN/control').read_text()
    assert 'libpam-systemd' in control and 'login' in control
    results['base'] = {'verified_regular_members': base_files, 'deb_sha256': file_sha(WORK / 'node-base.deb')}
    results['native'] = {'deb_sha256': file_sha(WORK / 'node-display.deb'), 'client_sha256': file_sha(WORK / 'libphoto-wall-frame-client.so')}
    (WORK / 'artifact-verification.json').write_text(json.dumps(results, sort_keys=True))
    print(json.dumps(results, sort_keys=True))

if __name__ == '__main__':
    main()
