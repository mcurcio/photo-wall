import json
import os
import subprocess
import sys
from psycopg.conninfo import make_conninfo
# Local Compose development DB; fixtures use random schemas and drop only those.
record = json.loads(subprocess.check_output(['docker','inspect','photo-wall-database-1']))[0]
settings = dict(item.split('=',1) for item in record['Config']['Env'] if '=' in item)
env = dict(os.environ)
env['PHOTO_WALL_TEST_DATABASE_URL'] = make_conninfo(
    host='127.0.0.1',port=54329,user=settings['POSTGRES_USER'],
    password=settings['POSTGRES_PASSWORD'],dbname=settings['POSTGRES_DB'])
raise SystemExit(subprocess.call([sys.executable,'-m','pytest','-q','-p','no:cacheprovider',*sys.argv[1:]],env=env))
