import hashlib,json,tarfile
from pathlib import Path
w=Path('/Volumes/Dock/Temp/photo-wall-node-resume-20260930/lifecycle-fixtures-stopfix')
def sha(stream):
 h=hashlib.sha256()
 while data:=stream.read(1048576):h.update(data)
 return h.hexdigest()
results={}
for role in ['success','failure']:
 ref=json.loads((w/(role+'-reference.json')).read_text());path=w/(role+'.tar')
 with path.open('rb') as f:assert sha(f)==ref['environment_sha256']
 assert path.stat().st_size==ref['size_bytes']
 with (w/(role+'.deb')).open('rb') as f:assert sha(f)==ref['deb_sha256']
 with tarfile.open(path) as t:
  meta=json.load(t.extractfile('environment.json'));actual={}
  assert meta['reference']=={k:v for k,v in ref.items() if k not in ['environment_sha256','size_bytes']}
  for m in t:
   if not m.name.startswith('rootfs/') or m.isdir():continue
   name=m.name[7:]
   if m.issym():actual[name]={'link':m.linkname}
   else:
    assert m.isfile();actual[name]={'mode':m.mode,'size':m.size,'sha256':sha(t.extractfile(m))}
  assert actual==meta['files']
  for filename,key in [('dependency-lock.json','dependency_lock_sha256'),('sources.json','source_snapshot_sha256')]:assert sha(t.extractfile(filename))==ref[key]
  staged=w/(role+'-package')
  count=0
  for p in (staged/'usr/lib/photo-wall-player').rglob('*'):
   if p.is_file():
    name=p.relative_to(staged).as_posix()
    with p.open('rb') as f:assert sha(f)==actual[name]['sha256']
    count+=1
  results[role]={'reference':ref,'verified_members':len(actual),'package_source_files':count,'fixture_only':True}
(w/'fixture-artifact-verification.json').write_text(json.dumps(results,sort_keys=True));print('VERIFIED', {k:v['verified_members'] for k,v in results.items()})
